# PF Lab 03 — Problems

## 1. Display Student Information Using Different Data Types

### Pseudocode

START

DECLARE studentName AS STRING
DECLARE rollNumber AS INTEGER
DECLARE age AS INTEGER
DECLARE percentage AS FLOAT
DECLARE grade AS CHARACTER

SET studentName = "Ghulam Mustafa"
SET rollNumber = 12345
SET age = 19
SET percentage = 88.5
SET grade = 'A'

DISPLAY studentName
DISPLAY rollNumber
DISPLAY age
DISPLAY percentage
DISPLAY grade

END


## 2. Read and Display a Character Using getchar() and putchar()

### Pseudocode

START

DECLARE character AS CHARACTER

DISPLAY "Enter a character: "
READ character using getchar()

DISPLAY "The character is: "
DISPLAY character using putchar()

END


## 3. Display a Floating-Point Value Using Different Precision Settings

### Pseudocode

START

DECLARE number AS FLOAT

SET number = 123.456789

DISPLAY number with default precision
DISPLAY number with 2 decimal places
DISPLAY number with 4 decimal places
DISPLAY number with 6 decimal places

END
